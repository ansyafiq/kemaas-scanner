  <script>
    let currentUser = null;
    let html5QrcodeScanner = null;
    let currentScannedData = null;

    // --- LOGOUT SOLUTION: KEMBALI KE LOGIN PAGE DENGAN BERSIH ---
    function logout() {
      if (html5QrcodeScanner) {
        stopScanner();
      }
      currentUser = null;
      currentScannedData = null;

      // Reset borang
      document.getElementById('loginForm').reset();
      document.getElementById('signupForm').reset();

      // Sembunyikan semua seksyen Dashboard
      document.getElementById('dashboardView').classList.add('d-none');
      document.getElementById('securitySection').classList.add('d-none');
      document.getElementById('wardenSection').classList.add('d-none');
      document.getElementById('studentSection').classList.add('d-none');
      document.getElementById('developerSection').classList.add('d-none');
      document.getElementById('scanResultModal').classList.add('d-none');

      // Tunjukkan semula halaman Log Masuk
      document.getElementById('authView').classList.remove('d-none');
      switchAuthTab('login');
    }

    function switchAuthTab(type) {
      if(type === 'login') {
        document.getElementById('loginForm').classList.remove('d-none');
        document.getElementById('signupForm').classList.add('d-none');
        document.getElementById('tabLoginBtn').classList.add('active');
        document.getElementById('tabSignupBtn').classList.remove('active');
      } else {
        document.getElementById('loginForm').classList.add('d-none');
        document.getElementById('signupForm').classList.remove('d-none');
        document.getElementById('tabLoginBtn').classList.remove('active');
        document.getElementById('tabSignupBtn').classList.add('active');
      }
    }

    function toggleStudentFields() {
      const role = document.getElementById('signRole').value;
      if(role === 'Student') {
        document.getElementById('studentFields').classList.remove('d-none');
      } else {
        document.getElementById('studentFields').classList.add('d-none');
      }
    }

    function handleSignUp(e) {
      e.preventDefault();
      const btn = document.getElementById('signupSubmitBtn');
      btn.disabled = true;
      btn.innerText = "Sila tunggu...";

      const data = {
        role: document.getElementById('signRole').value,
        fullName: document.getElementById('signName').value,
        icNumber: document.getElementById('signIC').value.trim(),
        matrixNo: document.getElementById('signMatrix').value.trim(),
        guardianName: document.getElementById('signGuardian').value,
        guardianPhone: document.getElementById('signGuardianPhone').value,
        phoneNo: document.getElementById('signPhone').value,
        email: document.getElementById('signEmail').value.trim(),
        password: document.getElementById('signPass').value
      };
      
      google.script.run.withSuccessHandler(res => {
        btn.disabled = false;
        btn.innerText = "Daftar";
        alert(res.message);
        if(res.success) switchAuthTab('login');
      }).registerUser(data);
    }

    function handleLogin(e) {
      e.preventDefault();
      const btn = document.getElementById('loginSubmitBtn');
      btn.disabled = true;
      btn.innerText = "Memproses...";

      const email = document.getElementById('loginEmail').value.trim();
      const pass = document.getElementById('loginPass').value;
      
      google.script.run.withSuccessHandler(res => {
        btn.disabled = false;
        btn.innerText = "Log Masuk";
        if(res.success) {
          currentUser = res.user;
          renderDashboard();
        } else {
          alert(res.message);
        }
      }).loginUser(email, pass);
    }

    function renderDashboard() {
      document.getElementById('authView').classList.add('d-none');
      document.getElementById('dashboardView').classList.remove('d-none');
      document.getElementById('userGreeting').innerText = currentUser.fullName;
      document.getElementById('userBadge').innerText = currentUser.role;

      if (currentUser.role === 'Security') {
        document.getElementById('securitySection').classList.remove('d-none');
      } else if (currentUser.role === 'Warden') {
        document.getElementById('wardenSection').classList.remove('d-none');
        loadLogs();
      } else if (currentUser.role === 'Student') {
        document.getElementById('studentSection').classList.remove('d-none');
        loadStudentLogs();
      } else if (currentUser.role === 'Developer') {
        document.getElementById('wardenSection').classList.remove('d-none');
        document.getElementById('developerSection').classList.remove('d-none');
        loadLogs();
        loadUsers();
      }
    }

    // --- CAMERA & BARCODE SCANNER SOLUTION ---
    function startScanner() {
      document.getElementById('scanResultModal').classList.add('d-none');

      // Konfigurasi imbasan yang menyokong Barcode 1D (Code 128 / EAN) & QR
      const config = {
        fps: 15,
        qrbox: { width: 300, height: 150 }, // Bentuk melintang sesuai untuk barcode kad
        aspectRatio: 1.0,
        experimentalFeatures: {
          useBarCodeDetectorIfSupported: true
        }
      };

      if (!html5QrcodeScanner) {
        html5QrcodeScanner = new Html5Qrcode("reader");
      }

      // Utamakan kamera belakang peranti (facingMode: environment)
      html5QrcodeScanner.start(
        { facingMode: "environment" },
        config,
        (decodedText) => {
          // Berjaya Imbas!
          stopScanner();
          onBarcodeScanned(decodedText);
        },
        (errorMessage) => {
          // Abaikan ralat semasa mencari barcode
        }
      ).then(() => {
        document.getElementById('startScanBtn').classList.add('d-none');
        document.getElementById('stopScanBtn').classList.remove('d-none');
      }).catch(err => {
        alert("Gagal mengakses kamera peranti. Pastikan anda memberikan kebenaran (Permission) akses kamera pada pelayar anda.\n\nRalat: " + err);
      });
    }

    function stopScanner() {
      if (html5QrcodeScanner && html5QrcodeScanner.isScanning) {
        html5QrcodeScanner.stop().then(() => {
          document.getElementById('startScanBtn').classList.remove('d-none');
          document.getElementById('stopScanBtn').classList.add('d-none');
        }).catch(err => console.error(err));
      }
    }

    function onBarcodeScanned(icNumber) {
      // Bersihkan teks daripada sebarang ruang
      icNumber = icNumber.trim();

      google.script.run.withSuccessHandler(res => {
        if(!res.success) {
          alert(res.message);
          return;
        }
        currentScannedData = res;
        document.getElementById('scanResultModal').classList.remove('d-none');
        
        let html = `<p class="mb-1"><strong>Nama:</strong> ${res.student.fullName}</p>
                    <p class="mb-1"><strong>No. Matrik:</strong> ${res.student.matrixNo}</p>
                    <p class="mb-1"><strong>No. IC:</strong> ${res.student.icNumber}</p>`;
        
        if(res.action === 'CHECK_OUT') {
          document.getElementById('scanTitle').innerText = "Proses KELUAR Pelajar";
          document.getElementById('checkOutActions').classList.remove('d-none');
          document.getElementById('checkInActions').classList.add('d-none');
        } else {
          document.getElementById('scanTitle').innerText = "Proses MASUK Pelajar";
          html += `<p class="mb-1 text-danger"><strong>Tempoh Di Luar:</strong> ${res.duration}</p>
                   <p class="mb-1"><strong>Tujuan:</strong> ${res.activeLog.destination}</p>`;
          document.getElementById('checkInActions').classList.remove('d-none');
          document.getElementById('checkOutActions').classList.add('d-none');
        }
        document.getElementById('scanDetails').innerHTML = html;
      }).scanStudentBarcode(icNumber);
    }

    function processCheckOut(type) {
      const dest = document.getElementById('destinationInput').value.trim();
      if(!dest) {
        alert("Sila masukkan tujuan/destinasi pelajar terlebih dahulu.");
        return;
      }

      const payload = {
        icNumber: currentScannedData.student.icNumber,
        studentName: currentScannedData.student.fullName,
        matrixNo: currentScannedData.student.matrixNo,
        type: type,
        destination: dest,
        securityId: currentUser.userId
      };

      google.script.run.withSuccessHandler(res => {
        alert(res.message);
        document.getElementById('scanResultModal').classList.add('d-none');
        document.getElementById('destinationInput').value = '';
      }).confirmCheckOut(payload);
    }

    function processCheckIn() {
      const payload = {
        logId: currentScannedData.activeLog.logId,
        duration: currentScannedData.duration,
        securityId: currentUser.userId
      };

      google.script.run.withSuccessHandler(res => {
        alert(res.message);
        document.getElementById('scanResultModal').classList.add('d-none');
      }).confirmCheckIn(payload);
    }

    function loadLogs() {
      google.script.run.withSuccessHandler(data => {
        let rows = '';
        if(data.length === 0) {
          rows = `<tr><td colspan="6" class="text-center text-muted">Tiada rekod keluar/masuk.</td></tr>`;
        } else {
          data.forEach(l => {
            rows += `<tr>
              <td><strong>${l.studentName}</strong><br><small class="text-muted">${l.matrixNo}</small></td>
              <td>${l.type}</td>
              <td>${l.timeOut}</td>
              <td>${l.timeIn}</td>
              <td>${l.duration}</td>
              <td><span class="badge ${l.status==='Outside'?'bg-warning text-dark':'bg-success'}">${l.status}</span></td>
            </tr>`;
          });
        }
        document.getElementById('logsBody').innerHTML = rows;
      }).getAllLogs();
    }

    function loadStudentLogs() {
      google.script.run.withSuccessHandler(data => {
        let rows = '';
        if(data.length === 0) {
          rows = `<tr><td colspan="5" class="text-center text-muted">Tiada rekod sejarah pergerakan.</td></tr>`;
        } else {
          data.forEach(l => {
            rows += `<tr>
              <td>${l.type}</td>
              <td>${l.timeOut}</td>
              <td>${l.timeIn}</td>
              <td>${l.duration}</td>
              <td><span class="badge ${l.status==='Outside'?'bg-warning text-dark':'bg-success'}">${l.status}</span></td>
            </tr>`;
          });
        }
        document.getElementById('studentLogsBody').innerHTML = rows;
      }).getStudentLogs(currentUser.icNumber);
    }

    function loadUsers() {
      google.script.run.withSuccessHandler(users => {
        let rows = '';
        users.forEach(u => {
          rows += `<tr>
            <td>${u.fullName}</td>
            <td>${u.icNumber} / ${u.matrixNo || '-'}</td>
            <td>${u.email}</td>
            <td><span class="badge bg-info text-dark">${u.role}</span></td>
            <td>
              <button class="btn btn-danger btn-sm" onclick="deleteUserAction('${u.userId}')">Padam</button>
            </td>
          </tr>`;
        });
        document.getElementById('usersBody').innerHTML = rows;
      }).getAllUsers();
    }

    function deleteUserAction(userId) {
      if(confirm("Adakah anda pasti mahu memadam akaun pengguna ini?")) {
        google.script.run.withSuccessHandler(res => {
          alert(res.message);
          loadUsers();
        }).deleteUser(userId);
      }
    }

    function printPDF() {
      window.print();
    }
